export function mulberry32(seed) {
  let a = seed >>> 0;
  return function () {
    a |= 0;
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

export function scaledAnsatz(n, layers, params) {
  let s = StateVector.zero(n);
  let idx = 0;
  for (let l = 0; l < layers; l++) {
    for (let q = 0; q < n; q++) {
      s = applySingleQubitGate(s, rotationY(params[idx++]), q);
      s = applySingleQubitGate(s, rotationZ(params[idx++]), q);
    }
    if (n > 1) {
      for (let q = 0; q < n; q++) s = applyCNOT(s, q, (q + 1) % n);
    }
  }
  return s;
}

export function numParams(n, layers) {
  return n * layers * 2;
}

export function scaledCost(n, layers, params, H) {
  return expectationValue(scaledAnsatz(n, layers, params), H).re;
}

export function gradientVarianceForN(n, samples, seed) {
  const layers = 4 * n;
  const P = numParams(n, layers);
  const H = tensorAll(Array.from({ length: n }, () => PAULI_Z));
  const rng = mulberry32(seed);
  const grads = [];
  for (let s = 0; s < samples; s++) {
    const params = Array.from({ length: P }, () => rng() * 2 * Math.PI);
    const paramsPlus = params.slice();
    const paramsMinus = params.slice();
    paramsPlus[0] += Math.PI / 2;
    paramsMinus[0] -= Math.PI / 2;
    const cPlus = scaledCost(n, layers, paramsPlus, H);
    const cMinus = scaledCost(n, layers, paramsMinus, H);
    grads.push((cPlus - cMinus) / 2);
  }
  const mean = grads.reduce((a, b) => a + b, 0) / grads.length;
  const variance = grads.reduce((a, b) => a + (b - mean) ** 2, 0) / grads.length;
  return { layers, numParams: P, mean, variance };
}

export function landscapeCurvesForN(n, seed) {
  const layers = 4 * n;
  const P = numParams(n, layers);
  const H = tensorAll(Array.from({ length: n }, () => PAULI_Z));
  const rng = mulberry32(seed);
  const thetaSteps = 32;
  const drawColors = ["brand", "accent", "muted", "warning"];
  return drawColors.map((color, d) => {
    const otherParams = Array.from({ length: P }, () => rng() * 2 * Math.PI);
    const points = Array.from({ length: thetaSteps + 1 }, (_, k) => {
      const theta = (k / thetaSteps) * 2 * Math.PI;
      const params = otherParams.slice();
      params[0] = theta;
      return { x: theta, y: scaledCost(n, layers, params, H) };
    });
    return { label: `draw ${d + 1}`, color, points };
  });
}

export function landscapeGridForN(n, seed, steps = 12) {
  const layers = 4 * n;
  const P = numParams(n, layers);
  const H = tensorAll(Array.from({ length: n }, () => PAULI_Z));
  const rng = mulberry32(seed);
  const base = Array.from({ length: P }, () => rng() * 2 * Math.PI);
  const grid = [];
  for (let i = 0; i < steps; i++) {
    const theta = (i / (steps - 1)) * 2 * Math.PI;
    const row = [];
    for (let j = 0; j < steps; j++) {
      const phi = (j / (steps - 1)) * 2 * Math.PI;
      const params = base.slice();
      params[0] = theta;
      params[1] = phi;
      row.push({ theta, phi, value: scaledCost(n, layers, params, H) });
    }
    grid.push(row);
  }
  return grid;
}

export function gridRange(grid) {
  let min = Infinity;
  let max = -Infinity;
  for (const row of grid) for (const cell of row) {
    if (cell.value < min) min = cell.value;
    if (cell.value > max) max = cell.value;
  }
  return max - min;
}

## Why the Toy Example Converged

VQE's worked toy example converged without incident: 80 pattern-search
iterations, monotonic descent throughout, agreement with the exact answer
to $10^{-12}$. That example used a single qubit. This lesson asks the
question the toy example was too small to expose: what happens to that
same training process as the ansatz scales up to real problem sizes?
The answer is one of the most important, and most numerically verifiable,
results in variational quantum computing.

## Mean Zero, Variance Exponentially Small

### The parameter-shift rule, derived

For a gate $U(\theta)=e^{-i\theta P/2}$ with $P$ Hermitian and
$P^2=I$ (true for every Pauli, hence for $R_y,R_z$), Taylor expansion of
the matrix exponential gives the exact identity $U(\theta)=\cos(\theta/2)
I-i\sin(\theta/2)P$. Any expectation value $C(\theta)=\langle\psi|
U(\theta)^\dagger HU(\theta)|\psi\rangle$ built from this gate expands,
via the double-angle identities, into a pure single-frequency trig
polynomial:

$$
C(\theta) = a + b\cos\theta + c\sin\theta
$$

for constants $a,b,c$ depending on everything *except* $\theta$. A short
direct check confirms the **parameter-shift rule**:

$$
\frac{C(\theta+\pi/2)-C(\theta-\pi/2)}2 = \frac{[b(-\sin\theta)+c\cos\theta]-[b\sin\theta-c\cos\theta]}2 = -b\sin\theta+c\cos\theta = C'(\theta)
$$

an **exact** gradient formula (not a finite-difference approximation)
using only two extra circuit evaluations, the tool this lesson's
numerics use below.

### The gradient's mean is exactly zero

If $\theta_i$ (the parameter being differentiated) is drawn uniformly
over a full period, $\theta_i\sim\text{Unif}[0,2\pi)$, independent of
every other parameter, then since $C'(\theta_i)=-b\sin\theta_i+c\cos
\theta_i$ with $b,c$ depending only on the *other* parameters:

$$
\mathbb E[C'(\theta_i)] = \mathbb E_{\text{others}}\Big[\mathbb E_{\theta_i}[-b\sin\theta_i+c\cos\theta_i \mid b,c]\Big] = \mathbb E_{\text{others}}\Big[-b\!\!\underbrace{\int_0^{2\pi}\!\!\sin\theta_i\,\frac{d\theta_i}{2\pi}}_{=0}\!\!+c\!\!\underbrace{\int_0^{2\pi}\!\!\cos\theta_i\,\frac{d\theta_i}{2\pi}}_{=0}\Big] = 0
$$

**exactly zero**, by the law of total expectation and the elementary
fact that $\sin$ and $\cos$ each integrate to zero over a full period,
for *any* joint distribution of the other parameters, no deep
theory required. This is the fully rigorous, elementary half of the
barren plateau story: on average, the gradient doesn't point anywhere.

### Why the variance also vanishes: exponentially in qubit count

Zero mean alone is not yet a problem: that's true even for a
perfectly well-behaved, easily trainable cost landscape (a bowl-shaped
function's gradient also averages to zero over a symmetric domain). The
real obstruction is that the gradient's *typical size* (its standard
deviation) shrinks toward zero as the number of qubits grows, so
gradient-based optimization has nothing to work with even locally. The
mechanism (McClean, Boixo, Smelyanskiy, Babbush & Neven 2018; the
complete argument uses Weingarten calculus over the Haar measure, which
is substantial technical machinery this lesson does not build) is
**concentration of measure**: a sufficiently
deep, sufficiently expressive parametrized circuit behaves, statistically,
like a Haar-random unitary applied to the initial state. For a
Haar-random pure state $|\psi\rangle$ in a $2^n$-dimensional Hilbert
space and a fixed traceless observable $H$ (a global Pauli string like
$Z^{\otimes n}$ is this kind of observable), Lévy's lemma (a
standard concentration-of-measure theorem for Lipschitz functions on a
high-dimensional sphere, itself a citable, standard result) gives

$$
\text{Var}_\psi[\langle\psi|H|\psi\rangle] = O\!\left(\frac1{2^n}\right)
$$

and because the parameter-shift gradient is itself a difference of two
such expectation values at nearby parameter points, its variance
inherits the same $O(2^{-n})$ scaling. **<Term id="barren-plateaus">This is the barren plateau</Term>**:
not merely that the average gradient is zero (derived exactly above, and
true even for well-behaved landscapes), but that its *typical
fluctuation size* collapses exponentially, so a random or naively-chosen
starting point is, with overwhelming probability, sitting on a
region so flat that no local optimizer can find a direction of descent
within any reasonable numerical precision.

## What a Flat Landscape Costs the Optimizer

The single-qubit VQE toy example converged so easily because
$2^1=2$ is nowhere near large enough for concentration of measure to
bite: the whole point of that lesson's ansatz being "fully expressive"
(reaching every point on the Bloch sphere) meant the cost landscape had
findable structure everywhere. Concentration of measure is a
*high-dimensional* phenomenon: it's the same reason a random point on a
high-dimensional sphere is, with overwhelming probability, very close
to the equator, no matter which "pole" you call zero. A random
parameter setting on a large, expressive enough variational circuit is,
in this sense, always "near the equator" of the cost landscape,
nowhere near a useful gradient, for purely geometric reasons that have
nothing to do with how good the classical optimizer is.

## Blaming the Optimizer for the Landscape

<Callout type="note">
**Expecting the exponential shrinkage to show up cleanly at *any*
circuit depth.** This lesson's own verification process ran into this
directly: using only $n$ layers (rather than the $4n$ layers used
above), the measured variance at $n=2\to3$ *increased* before
decreasing. The increase is reproducible, and worth reporting. The
concentration-of-measure argument requires the circuit to be
"sufficiently random," which in practice means sufficiently *deep*; a
shallow ansatz hasn't yet scrambled enough to approximate a Haar-random
unitary, so the asymptotic trend can be non-monotonic at small
$n$ and shallow depth, only becoming clean once the depth scales
appropriately with $n$, as it does in the deeper ($4n$-layer) numbers
used above.
</Callout>

<Callout type="note">
**Concluding that VQE is therefore useless at any real qubit count.**
Barren plateaus specifically afflict circuits that are *both* deep
enough to be "sufficiently random" *and* trained against a *global*
cost function like $Z^{\otimes n}$. Shallower, problem-tailored ansätze
(not generic random circuits) and *local* cost functions (few-qubit
terms in place of an all-qubit observable) are active,
partially effective mitigation strategies in real research; the
phenomenon derived here is an obstacle for the generic case, not
a no-go theorem for every possible variational circuit design.
</Callout>

## Practice Questions

1. Using the exact identity $C(\theta)=a+b\cos\theta+c\sin\theta$,
   verify directly (by substitution) that $C(\theta+\pi/2)-C(\theta-\pi/2)$
   equals $-2b\sin\theta+2c\cos\theta$, confirming the parameter-shift
   rule's derivation above.

   <details className="answer-reveal">
   <summary>Answer</summary>

   Shift the argument in each trig term, using
   $\cos(\theta\pm\pi/2)=\mp\sin\theta$ and
   $\sin(\theta\pm\pi/2)=\pm\cos\theta$:

   $$
   C(\theta+\tfrac\pi2)=a-b\sin\theta+c\cos\theta,
   \qquad
   C(\theta-\tfrac\pi2)=a+b\sin\theta-c\cos\theta.
   $$

   Subtracting, the constant $a$ cancels and each remaining term doubles:

   $$
   C(\theta+\tfrac\pi2)-C(\theta-\tfrac\pi2)=-2b\sin\theta+2c\cos\theta=2\,C'(\theta),
   $$

   since $C'(\theta)=-b\sin\theta+c\cos\theta$ by direct differentiation.
   Dividing by two gives the rule as stated.

   Two features of that calculation deserve to be pointed at, because they
   are what make the rule useful rather than merely correct.

   The constant $a$ dropping out is not cosmetic. It holds *everything*
   the circuit does that does not involve $\theta$: every other parameter,
   the whole rest of the ansatz, the observable. None of it needs to be
   known, and none of it needs to be small.

   And the shift is $\pi/2$, an enormous step by finite-difference
   standards, yet the answer is the exact derivative rather than an
   approximation with an $O(h^2)$ truncation error. That is possible only
   because $C$ is a pure single-frequency trig polynomial: two samples
   determine a function with two unknown Fourier coefficients, exactly.
   There is no step-size tradeoff and no cancellation error from
   subtracting nearly equal numbers, which matters a great deal when each
   evaluation carries shot noise.

   The hypothesis making all of this work is $P^2=I$, so that
   $e^{-i\theta P/2}=\cos(\theta/2)I-i\sin(\theta/2)P$ and only the
   frequencies $0,\pm1$ appear. Break it and the rule breaks with it: a
   generator with three or more distinct eigenvalues produces additional
   harmonics, and a two-term shift rule can no longer determine them. Such
   gates need a longer multi-term shift rule with more evaluations, which
   is why "the parameter-shift rule" is really a statement about Pauli
   rotations rather than about parametrized gates in general.

   </details>

2. Explain why the zero-mean gradient result derived in this lesson
   requires *no* assumption about circuit depth or qubit count, while
   the vanishing-variance result specifically requires both "sufficiently
   deep" and "sufficiently many qubits."

   <details className="answer-reveal">
   <summary>Answer</summary>

   Because the zero-mean statement is an identity about a periodic
   function, while the variance statement is a claim about how a
   high-dimensional random object concentrates.

   **The mean.** There is a one-line version that makes its triviality
   plain. $C$ is $2\pi$-periodic in $\theta_i$, so

   $$
   \mathbb E[C'(\theta_i)]=\frac{1}{2\pi}\int_0^{2\pi}C'(\theta_i)\,d\theta_i=\frac{C(2\pi)-C(0)}{2\pi}=0
   $$

   by the fundamental theorem of calculus. Nothing about circuits appears:
   any periodic cost, at any depth, on any number of qubits, differentiated
   with respect to a uniformly distributed periodic parameter, has mean
   gradient zero. The lesson's derivation via $-b\sin+c\cos$ is the same
   fact with the structure exhibited.

   This is why the zero mean, on its own, says nothing about trainability.
   A perfect parabola in disguise would satisfy it too. It rules out one
   trivial failure mode, a systematic drift, and no more.

   **The variance.** The claim $\mathrm{Var}=O(2^{-n})$ is a
   concentration-of-measure result, and concentration needs two separate
   things to be true.

   The dimension has to be large. Lévy's lemma bounds fluctuations of a
   Lipschitz function on the unit sphere in $\mathbb C^{2^n}$ by something
   shrinking like $2^{-n}$. At $n=1$ that "sphere" is the Bloch sphere, and
   an expectation value on it swings across the full $[-1,1]$ range with no
   concentration whatsoever. This is exactly why the single-qubit VQE toy
   example trained without incident.

   The circuit has to be random enough for the lemma to apply. The state
   $U(\theta)|\psi_0\rangle$ is not Haar-distributed for a fixed shallow
   ansatz; it is confined to a low-dimensional manifold that the ansatz can
   reach. Only once the circuit is deep enough to scramble, approximating
   at least a unitary $2$-design, does it behave statistically like a Haar
   draw and inherit the bound. The lesson's own callout is the evidence:
   at $n$ layers rather than $4n$, the measured variance *increased* from
   $n=2$ to $n=3$ before turning over. A shallow circuit has not earned the
   theorem yet.

   The asymmetry in logical type is worth naming as well. The mean result
   is exact at every finite $n$ and every depth. The variance result is
   asymptotic, about a family of circuits as $n$ grows, and the measured
   ratios $0.53,0.39,0.62,0.45$ scattering around $1/2$ are what an
   asymptotic claim looks like when sampled $300$ times at small $n$.

   </details>

3. From the measured variance ratios ($n=2\to6$), estimate (order of
   magnitude) what the gradient variance might be at $n=20$ qubits if
   the same roughly-halving-per-qubit trend continued, and explain why
   this makes gradient-based training hopeless at that scale
   without a fundamentally different ansatz or cost function design.

   <details className="answer-reveal">
   <summary>Answer</summary>

   About $5\times10^{-7}$, so a typical gradient component of roughly
   $7\times10^{-4}$. The reason that is fatal is not that the number is
   small but that resolving it costs a number of measurement shots growing
   like $2^n$.

   The extrapolation is a factor of $2$ per qubit over $14$ more qubits
   from the measured endpoint:

   $$
   \mathrm{Var}(n{=}20)\approx7.557\times10^{-3}\times2^{-14}=4.6\times10^{-7},
   \qquad
   \sigma\approx6.8\times10^{-4}.
   $$

   The measured data supports the exponent well enough for an
   order-of-magnitude claim: from $n=2$ to $n=6$ the variance fell by
   $0.1301/0.007557=17.2$, against $2^4=16$ predicted.

   Now the part that matters. A gradient of typical size $\sigma$ has to be
   distinguished from zero, and every circuit evaluation of
   $\langle Z^{\otimes n}\rangle$ is an average of $\pm1$ outcomes, so $S$
   shots give it a standard error of about $1/\sqrt S$. The parameter-shift
   gradient combines two such evaluations, with shot noise roughly
   $1/\sqrt{2S}$. Requiring that to fall below $\sigma$:

   $$
   S\gtrsim\frac{1}{2\sigma^2}=\frac{1}{2\,\mathrm{Var}}\approx1.1\times10^{6}
   \ \text{shots per shifted circuit}.
   $$

   The ansatz at $n=20$ carries $8n^2=3200$ parameters, each needing two
   shifted evaluations, so a single gradient step costs on the order of
   $3200\times2\times10^{6}\approx6\times10^{9}$ circuit executions. That
   is one step of one optimization run.

   And the scaling is the whole point: shots per component grow like
   $1/\mathrm{Var}\sim2^{n}$. The measurement cost of *reading* the
   gradient is exponential in the qubit count, which cancels precisely
   whatever exponential advantage the circuit was supposed to provide. At
   $n=50$ the extrapolation gives $\mathrm{Var}\approx4\times10^{-16}$ and
   $S\approx10^{15}$ shots per evaluation, which is not a hard engineering
   problem but an impossible one.

   Nothing about the classical optimizer enters this. Adam, BFGS, a
   natural-gradient method: all of them are being handed a number that is
   noise. The fixes have to change the landscape rather than the search,
   which is why the active research directions are shallower
   problem-tailored ansätze, local cost functions, and clever
   initialization near a known-good point, all of which attack the variance
   rather than the descent rule.

   </details>

4. Why does using a *global* observable like $Z^{\otimes n}$ (rather
   than, say, just $Z$ on one qubit) make the barren plateau problem
   worse? (Hint: think about how many of the $2^n$ computational basis
   states have a given eigenvalue under each observable.)

   <details className="answer-reveal">
   <summary>Answer</summary>

   Because $Z^{\otimes n}$ asks a question that every one of the $2^n$
   basis states participates in equally, so its expectation value is a sum
   of $2^n$ terms that almost perfectly cancel, while a single-qubit $Z$
   only ever probes the few qubits the circuit has had time to connect to
   it.

   Take the hint literally first. $Z^{\otimes n}$ has just two eigenvalues,
   $\pm1$, each with degeneracy $2^{n-1}$: it reads the global parity of
   the bit string and nothing else. For a state spread over many basis
   strings,

   $$
   \langle Z^{\otimes n}\rangle=\sum_{x:\,\text{even}}|c_x|^2-\sum_{x:\,\text{odd}}|c_x|^2
   $$

   is a difference between two halves of the distribution, each of total
   weight near $\tfrac12$. Getting a value away from zero requires the
   amplitudes to be correlated with parity across all $n$ qubits at once,
   which is a vanishingly special condition on a spread-out state. Single
   $Z_1$, by contrast, splits the basis the same way but according to one
   bit, and that bit's marginal is something a shallow circuit can still
   control.

   The sharper mechanism is the light cone, and it is what makes locality
   a *depth-dependent* remedy rather than a universal one. Differentiating
   with respect to $\theta_i$ produces an expression involving only the
   part of the circuit lying between that gate and the observable. For
   $Z_1$ in an architecture with local entangling gates, that region
   touches only $O(d)$ qubits at depth $d$, so the relevant Hilbert space
   has effective dimension $2^{O(d)}$ rather than $2^n$, and concentration
   of measure is correspondingly weaker. For $Z^{\otimes n}$ the
   observable's support is all $n$ qubits from the outset, so the effective
   dimension is $2^n$ immediately and the concentration is exponential in
   $n$ at any depth at all.

   This is the content of the cost-function-dependent barren plateau
   result (Cerezo, Sone, Volkoff, Cincio and Coles, 2021): for a global
   cost the gradient variance is exponentially small even for
   constant-depth circuits, while for a local cost and depth $O(\log n)$
   it decays only polynomially in $n$, which is trainable. This lesson's
   own numbers sit squarely in the bad regime, using $Z^{\otimes n}$ with
   $4n$ layers.

   One caution against over-reading the fix. Locality buys trainability
   only while the circuit stays shallow. Push a local-cost ansatz to depth
   well beyond $O(\log n)$ and the light cone grows to cover everything,
   the circuit approaches a $2$-design, and the plateau returns. Locality
   and shallowness are one mitigation, not two, and neither survives the
   other's absence.

   </details>

## Summary

- The parameter-shift rule gives an *exact* gradient formula for
  $R_y$/$R_z$-based ansätze, derived directly from the single-frequency
  trig-polynomial structure of $C(\theta)$.
- The gradient's mean is exactly zero over uniformly random parameters,
  a short, fully elementary, rigorous derivation requiring no deep
  theory.
- The gradient's *variance* shrinks as $O(2^{-n})$ for sufficiently deep,
  sufficiently random ansätze and global cost functions, via
  concentration of measure (Lévy's lemma / McClean et al. 2018, cited
  rather than fully re-derived); this, not the zero mean, is the real
  barren plateau.
- Verified directly: scaling VQE's own ansatz pattern from $2$ to $6$
  qubits (using this platform's real gate and expectation-value engine,
  a deterministic seeded random generator, and the exact parameter-shift
  formula derived above) shows measured gradient variance shrinking by
  roughly half at each additional qubit, a $\sim17\times$ drop
  from $n=2$ to $n=6$, and, honestly reported, the trend is only clean
  once the ansatz is deep enough, as the "sufficiently random"
  precondition predicts.
